function getLateCancellations(type) {


var m = metas()[type];


return (state.sessions[type] || []).filter(function (c) {


return c.cancelled && c.cancelledTs && c.cancelledTs > m.cancelDeadline.getTime();


}).sort(function (a, b) { return b.cancelledTs - a.cancelledTs; });


}

}

loadResults: function () {


var type = state.resultsType;


var occ = pastOccurrences(type, 1)[0];


state.resultsOccurrence = occ; state.resultsCourts = null; state.resultsSavedAt = null; render();


Promise.all([


fetchConfirmations(type, occ.key),


sb.from("results").select("*").eq("session_type", type).eq("session_date", occ.key)


]).then(function (r) {


var confirmations = r[0].filter(function (c) { return !c.cancelled; }).sort(function (a, b) { return a.ts - b.ts; }).slice(0, capacityFor(type));


var courts = buildCourts(confirmations);


var savedRows = (r[1] && r[1].data) || [];


if (savedRows.length) {


var byCourt = {};


savedRows.forEach(function (row) {


byCourt[row.court_index] = byCourt[row.court_index] || {};


byCourt[row.court_index][row.player_id] = row;


});


courts.forEach(function (court, ci) {


court.forEach(function (p) {


var row = byCourt[ci] && byCourt[ci][p.playerId];


if (row) p.position = row.position;


});


});


state.resultsSavedAt = new Date(savedRows[0].saved_at).getTime();


}


state.resultsCourts = courts; render();


});


},

var next = [1, 2, 3, 4].filter(function (n) { return used.indexOf(n) === -1; })[0];


player.position = next || null;


}


render();

var rows = [];

session_type: type, session_date: occ.key, court_index: ci,


player_id: p.playerId, player_name: p.name, position: p.position,


saved_at: new Date().toISOString()


});


});


});

state.resultsSavedAt = Date.now();


fetchRanking().then(function (r) { state.ranking = r; render(); });


});


});
